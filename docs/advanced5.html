応用5: BGMと効果音

音を追加してゲームをもっと楽しくしよう！

この応用で学ぶこと

  • ✓ AudioSourceとAudioClipの使い方
  • ✓ BGM（ループ再生）と効果音（1回再生）の違い
  • ✓ シングルトンで音を一元管理する方法
  • ✓ PlayOneShot()で複数の音を重ねて再生

仕組みを考えよう

BGM（バックグラウンドミュージック）

  • ・ゲーム中ずっと流れる音楽
  • ・loop = true で繰り返し再生
  • ・AudioSourceのPlay()で開始
  • ・シーンごとに違うBGMを流せる

効果音（SE）

  • ・アクション時に1回だけ鳴る音
  • ・PlayOneShot()で再生
  • ・同時に複数鳴らせる
  • ・例: ジャンプ、アイテム取得、ゲームオーバー

ポイント: SoundManagerというシングルトンを作り、どこからでも SoundManager.Instance.PlaySE("jump") のように呼び出せるようにします。GameManagerと同じパターンです！

音素材の準備

注意: 音素材はフリー素材サイトからダウンロードしましょう。「フリー 効果音」で検索すると見つかります。.wav や .mp3 ファイルをAssetsフォルダにドラッグするだけでUnityに取り込めます。

用意する音素材

SoundManager.cs（新規作成） 

using UnityEngine;

/// <summary>
/// BGMと効果音を管理するクラス
/// EmptyObjectにアタッチして使用
/// </summary>
public class SoundManager : MonoBehaviour
{
    // シングルトンインスタンス
    public static SoundManager Instance { get; private set; }

    [Header("BGM")]
    [SerializeField]
    private AudioClip bgmTitle;       // タイトルBGM

    [SerializeField]
    private AudioClip bgmGame;        // ゲームBGM

    [Header("効果音")]
    [SerializeField]
    private AudioClip seJump;         // ジャンプ音

    [SerializeField]
    private AudioClip seItem;         // アイテム取得音

    [SerializeField]
    private AudioClip seStomp;        // 踏みつけ音

    [SerializeField]
    private AudioClip seGameOver;     // ゲームオーバー音

    [SerializeField]
    private AudioClip seClear;        // クリア音

    [Header("音量設定")]
    [SerializeField]
    private float bgmVolume = 0.5f;   // BGM音量

    [SerializeField]
    private float seVolume = 1.0f;    // 効果音音量

    // AudioSourceコンポーネント
    private AudioSource bgmSource;    // BGM用
    private AudioSource seSource;     // 効果音用

    void Awake()
    {
        // シングルトンパターン（GameManagerと同じ！）
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
            SetupAudioSources();
        }
        else
        {
            Destroy(gameObject);
        }
    }

    /// <summary>
    /// AudioSourceを2つ作成（BGM用と効果音用）
    /// </summary>
    private void SetupAudioSources()
    {
        // BGM用のAudioSource
        bgmSource = gameObject.AddComponent<AudioSource>();
        bgmSource.loop = true;        // ループ再生
        bgmSource.volume = bgmVolume;
        bgmSource.playOnAwake = false;

        // 効果音用のAudioSource
        seSource = gameObject.AddComponent<AudioSource>();
        seSource.loop = false;        // ループしない
        seSource.volume = seVolume;
        seSource.playOnAwake = false;
    }

    /// <summary>
    /// BGMを再生する
    /// </summary>
    public void PlayBGM(string bgmName)
    {
        AudioClip clip = null;

        switch (bgmName)
        {
            case "title":
                clip = bgmTitle;
                break;
            case "game":
                clip = bgmGame;
                break;
        }

        if (clip != null && bgmSource.clip != clip)
        {
            bgmSource.clip = clip;
            bgmSource.Play();
        }
    }

    /// <summary>
    /// BGMを停止する
    /// </summary>
    public void StopBGM()
    {
        bgmSource.Stop();
    }

    /// <summary>
    /// 効果音を再生する
    /// </summary>
    public void PlaySE(string seName)
    {
        AudioClip clip = null;

        switch (seName)
        {
            case "jump":
                clip = seJump;
                break;
            case "item":
                clip = seItem;
                break;
            case "stomp":
                clip = seStomp;
                break;
            case "gameover":
                clip = seGameOver;
                break;
            case "clear":
                clip = seClear;
                break;
        }

        if (clip != null)
        {
            // PlayOneShotは同時に複数の音を鳴らせる
            seSource.PlayOneShot(clip, seVolume);
        }
    }
}

private void HandleJump()
{
    if (Keyboard.current != null &&
        Keyboard.current.upArrowKey.wasPressedThisFrame &&
        isGrounded)
    {
        rb.linearVelocity = new Vector2(rb.linearVelocity.x, jumpForce);

        // ジャンプ音を鳴らす ← 追加！
        if (SoundManager.Instance != null)
        {
            SoundManager.Instance.PlaySE("jump");
        }
    }
}

public void CollectItem()
{
    itemCount++;

    // アイテム取得音 ← 追加！
    if (SoundManager.Instance != null)
    {
        SoundManager.Instance.PlaySE("item");
    }

    if (itemCount >= requiredItemCount)
    {
        GameClear();
    }
}

GameManager.cs - GameOver() / GameClear() 

public void GameOver()
{
    CurrentState = GameState.GameOver;

    // BGM停止 & ゲームオーバー音 ← 追加！
    if (SoundManager.Instance != null)
    {
        SoundManager.Instance.StopBGM();
        SoundManager.Instance.PlaySE("gameover");
    }

    SceneManager.LoadScene("GameOverScene");
}

public void GameClear()
{
    CurrentState = GameState.GameClear;

    // BGM停止 & クリア音 ← 追加！
    if (SoundManager.Instance != null)
    {
        SoundManager.Instance.StopBGM();
        SoundManager.Instance.PlaySE("clear");
    }

    SceneManager.LoadScene("GameClearScene");
}

GameManager.cs - StartGame() / UpdateStateFromScene() 

public void StartGame()
{
    itemCount = 0;
    CurrentState = GameState.Playing;

    // ゲームBGMを再生 ← 追加！
    if (SoundManager.Instance != null)
    {
        SoundManager.Instance.PlayBGM("game");
    }

    SceneManager.LoadScene("GameScene");
}

private void UpdateStateFromScene()
{
    string sceneName = SceneManager.GetActiveScene().name;
    // ... 既存の処理 ...

    // タイトルならタイトルBGMを再生 ← 追加！
    if (sceneName == "TitleScene" &&
        SoundManager.Instance != null)
    {
        SoundManager.Instance.PlayBGM("title");
    }
}

解説

  • ・AudioSource は音を再生するためのコンポーネントです。BGM用と効果音用の2つを作ります
  • ・loop = true にするとBGMが繰り返し再生されます
  • ・PlayOneShot() は効果音の再生に使います。同時に複数の音を重ねられます
  • ・bgmSource.clip != clip のチェックで、同じBGMが二重再生されるのを防ぎます
  • ・GameManagerと同じシングルトンなので、どのスクリプトからでも呼び出せます
